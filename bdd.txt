-------------------------------------------------------
-- TABLE GZONE
-------------------------------------------------------
CREATE TABLE GZONE (
    ID_ZONE NUMBER PRIMARY KEY,
    POPULATION NUMBER,
    X NUMBER,
    Y NUMBER,
    L NUMBER,
    H NUMBER,
    SURFACE NUMBER,
    DENSITE NUMBER
);

-------------------------------------------------------
-- TABLE GESPUB
-------------------------------------------------------
CREATE TABLE GESPUB (
    ID_BIN NUMBER PRIMARY KEY,
    CAPACITE FLOAT,
    STATUS VARCHAR2(50),
    ID_ZONE NUMBER,
    CONSTRAINT FK_GESPUB_ZONE FOREIGN KEY (ID_ZONE)
        REFERENCES GZONE(ID_ZONE) ON DELETE CASCADE
);

-------------------------------------------------------
-- TABLE HABITANT
-------------------------------------------------------
CREATE TABLE HABITANT (
    ID_HABITANT NUMBER PRIMARY KEY,
    NOM VARCHAR2(50) NOT NULL,
    PRENOM VARCHAR2(50) NOT NULL,
    ADRESSE VARCHAR2(100),
    ZONE_RESIDENCE VARCHAR2(50),
    SITUATION_FAMILIALE VARCHAR2(50),
    CONTACT VARCHAR2(30),
    STATUT VARCHAR2(20),
    BESOINS_SPECIFIQUES VARCHAR2(200),
    NIVEAU_SATISFACTION VARCHAR2(20),
    CONSTRAINT CK_STATUT_HABITANT CHECK (STATUT IN ('Actif','Inactif'))
);

CREATE INDEX IDX_HABITANT_ZONE ON HABITANT(ZONE_RESIDENCE);

-------------------------------------------------------
-- TABLE HABITAT
-------------------------------------------------------
CREATE TABLE HABITAT (
    ID_HABITAT NUMBER PRIMARY KEY,
    TYPE_LOGEMENT VARCHAR2(50),
    SURFACE NUMBER(10,2),
    ADRESSE VARCHAR2(255),
    NB_PERSONNE NUMBER,
    ID_ZONE NUMBER,
    CONSTRAINT FK_HABITAT_ZONE FOREIGN KEY (ID_ZONE)
        REFERENCES GZONE(ID_ZONE)
);

-------------------------------------------------------
-- SEQUENCE + TRIGGER FOR HABITAT
-------------------------------------------------------
CREATE SEQUENCE HABITAT_SEQ START WITH 1 INCREMENT BY 1;

CREATE OR REPLACE TRIGGER TRG_HABITAT_ID
BEFORE INSERT ON HABITAT
FOR EACH ROW
BEGIN
  IF :NEW.ID_HABITAT IS NULL THEN
    SELECT HABITAT_SEQ.NEXTVAL INTO :NEW.ID_HABITAT FROM dual;
  END IF;
END;
/

-------------------------------------------------------
-- TABLE PERSONNEL
-------------------------------------------------------
CREATE TABLE PERSONNEL (
    CIN NUMBER PRIMARY KEY,
    NOM VARCHAR2(50) NOT NULL,
    PRENOM VARCHAR2(50) NOT NULL,
    ZONE_AFFECTATION VARCHAR2(50),
    DISPONIBILITE VARCHAR2(30),
    COMPETENCES VARCHAR2(100),
    CONSTRAINT CK_DISPO_PERSONNEL CHECK
        (DISPONIBILITE IN ('Disponible','Indisponible'))
);

CREATE INDEX IDX_PERSONNEL_ZONE ON PERSONNEL(ZONE_AFFECTATION);

-------------------------------------------------------
-- TABLE VEHICULE
-------------------------------------------------------
CREATE TABLE VEHICULE (
    ID_VEHICULE NUMBER PRIMARY KEY,
    TYPE_VEHICULE VARCHAR2(50),
    MATRICULE VARCHAR2(20),
    CAPACITE VARCHAR2(20),
    ITINERAIRE VARCHAR2(150),
    HEURE_DEPART VARCHAR2(20),
    HEURE_ARRIVEE VARCHAR2(20),
    STATUT VARCHAR2(30),
    ZONE VARCHAR2(50),
    ID_HABITANT NUMBER,
    CONSTRAINT CK_TYPE_VEHICULE CHECK (TYPE_VEHICULE IN ('Bus','Tram','Électrique','Voiture')),
    CONSTRAINT CK_STATUT_VEHICULE CHECK (STATUT IN ('En service','En recharge','En panne','Disponible')),
    CONSTRAINT FK_VEHICULE_HABITANT FOREIGN KEY (ID_HABITANT)
        REFERENCES HABITANT(ID_HABITANT)
);

CREATE INDEX IDX_VEHICULE_HABITANT ON VEHICULE(ID_HABITANT);
CREATE INDEX IDX_VEHICULE_ZONE ON VEHICULE(ZONE);

-------------------------------------------------------
-- SEQUENCE + TRIGGER FOR VEHICULE
-------------------------------------------------------
CREATE SEQUENCE VEHICULE_SEQ START WITH 1 INCREMENT BY 1;

CREATE OR REPLACE TRIGGER VEHICULE_BI
BEFORE INSERT ON VEHICULE
FOR EACH ROW
BEGIN
    SELECT VEHICULE_SEQ.NEXTVAL INTO :NEW.ID_VEHICULE FROM dual;
END;
/

-------------------------------------------------------
-- TABLE UTILISATION_VEHICULE
-------------------------------------------------------
CREATE TABLE UTILISATION_VEHICULE (
    ID_UTILISATION NUMBER PRIMARY KEY,
    ID_HABITANT NUMBER NOT NULL,
    ID_VEHICULE NUMBER NOT NULL,
    DATE_UTILISATION DATE DEFAULT SYSDATE,
    HEURE_DEBUT VARCHAR2(20),
    HEURE_FIN VARCHAR2(20),
    CONSTRAINT FK_UTIL_HABITANT FOREIGN KEY (ID_HABITANT)
        REFERENCES HABITANT(ID_HABITANT),
    CONSTRAINT FK_UTIL_VEHICULE FOREIGN KEY (ID_VEHICULE)
        REFERENCES VEHICULE(ID_VEHICULE)
);

CREATE INDEX IDX_UTILISATION_HABITANT ON UTILISATION_VEHICULE(ID_HABITANT);
CREATE INDEX IDX_UTILISATION_VEHICULE ON UTILISATION_VEHICULE(ID_VEHICULE);
